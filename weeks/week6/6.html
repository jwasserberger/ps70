// === OLED (SSD1306 I2C) Setup ===
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1  // not used
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// === BLE Keyboard ===
BleKeyboard bleKeyboard("CapKey", "YourName", 100);

// === Mode Configuration ===
enum Mode {
  COLD_EMAIL = 0,
  CALL_FOLLOWUP = 1,
  CUSTOM = 2,
  MODE_COUNT = 3
};
const char* modeNames[MODE_COUNT] = {
  "Cold Email",
  "Call Follow-Up",
  "Custom"
};

// Pre-written text blocks for each mode
const char* textBlocks[MODE_COUNT][3] = {
  {
    "I hope this email finds you well. I came across your company and was impressed by your work in the industry.",
    "I'd love to schedule a quick call to discuss how our solutions might help address some of the challenges you're facing.",
    "Looking forward to your response. Thank you for your time."
  },
  {
    "Thank you for taking the time to speak with me today. I appreciate your insights and feedback on our discussion.",
    "As promised, I've attached the information we discussed during our call. Please let me know if you have any questions.",
    "I'll follow up next week to discuss next steps. Have a great day!"
  },
  {
    "Custom text block 1 - You can customize this in the code.",
    "Custom text block 2 - You can customize this in the code.",
    "Custom text block 3 - You can customize this in the code."
  }
};

// === Touch Key Config ===
struct Key {
  uint8_t       pin;
  char          character;
  unsigned long lastPressed;
};
const int numKeys = 4;
Key keys[numKeys] = {
  { 2,  'A', 0 },  // Text block 1
  { 4,  'B', 0 },  // Text block 2
  {13,  'C', 0 },  // Text block 3
  {15,  'D', 0 }   // Mode switch
};

const int touchThreshold = 20;   // capacitive threshold
const int debounceTime    = 300; // ms
Mode currentMode = COLD_EMAIL;

// Typing variables
bool    isSendingText   = false;
String  currentText     = "";
int     currentPosition = 0;
unsigned long nextTypeTime = 0;
const int batchSize  = 5;
const int batchPause = 40;

// BLE connection monitoring
unsigned long lastConnectionCheck = 0;
bool         wasConnected         = false;
unsigned long disconnectTime      = 0;
const int    reconnectDelay       = 1000;

void setup() {
  Serial.begin(115200);

  // OLED init
  Wire.begin(21, 22);
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("SSD1306 init failed");
    while (1);
  }
  display.clearDisplay();
  display.setRotation(2);
  display.setTextSize(2);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0,0);
  display.println("Starting BLE...");
  display.display();

  // BLE init
  bleKeyboard.begin();
  Serial.println("BLE started, waiting for connection...");
  updateModeDisplay();
  delay(1000);
}

void loop() {
  // --- BLE connection monitor ---
  if (millis() - lastConnectionCheck > 500) {
    bool isConn = bleKeyboard.isConnected();
    if (isConn != wasConnected) {
      wasConnected = isConn;
      if (isConn) {
        Serial.println("BLE Connected!");
        updateModeDisplay();
      } else {
        Serial.println("BLE Disconnected");
        disconnectTime = millis();
        display.fillScreen(SSD1306_BLACK);
        display.setTextSize(2);
        display.setCursor(0,0);
        display.println("BLE");
        display.println("Disconnected");
        display.display();
        isSendingText = false;
      }
    }
    lastConnectionCheck = millis();
  }

  // --- Sending text in batches ---
  if (isSendingText && bleKeyboard.isConnected()) {
    unsigned long now = millis();
    if (now >= nextTypeTime) {
      int cnt = 0;
      while (currentPosition < currentText.length() && cnt < batchSize) {
        bleKeyboard.write(currentText.charAt(currentPosition++));
        cnt++;
        delay(15);
      }
      int pct = (currentPosition * 100) / currentText.length();
      updateSendingDisplay(pct);
      if (currentPosition >= currentText.length()) {
        isSendingText = false;
        delay(300);
        updateModeDisplay();
      } else {
        nextTypeTime = now + batchPause;
      }
    }
  }
  // --- Otherwise handle capacitive touches ---
  else if (bleKeyboard.isConnected()) {
    for (int i = 0; i < numKeys; i++) {
      int val = touchRead(keys[i].pin);
      if (val < touchThreshold && millis() - keys[i].lastPressed > debounceTime) {
        keys[i].lastPressed = millis();
        if (i == numKeys - 1) {
          switchMode();
        } else {
          startSendingTextBlock(i);
        }
      }
    }
  }
  // --- Reconnect logic ---
  else if (!wasConnected && millis() - disconnectTime > reconnectDelay) {
    display.fillScreen(SSD1306_BLACK);
    display.setTextSize(1);
    display.setCursor(0,0);
    display.println("Reconnecting...");
    display.display();
    static int attempts = 0;
    if (++attempts > 5) {
      Serial.println("Restarting BLE");
      bleKeyboard.end();
      delay(500);
      bleKeyboard.begin();
      attempts = 0;
    }
    disconnectTime = millis();
  }

  delay(10);
}

void updateModeDisplay() {
  display.fillScreen(SSD1306_BLACK);
  display.setTextSize(2);
  display.setCursor(0,0);
  display.println(modeNames[currentMode]);
  display.display();
}

void updateSendingDisplay(int pct) {
  display.fillScreen(SSD1306_BLACK);
  display.setTextSize(2);
  display.setCursor(0,0);
  display.println("Sending...");
  display.drawRect(0,40,126,10,SSD1306_WHITE);
  display.fillRect(1,41,(pct*124)/100,8,SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(50,54);
  display.print(pct);
  display.print("%");
  display.display();
}

void switchMode() {
  currentMode = (Mode)((currentMode + 1) % MODE_COUNT);
  updateModeDisplay();
  Serial.printf("Mode: %s\n", modeNames[currentMode]);
  delay(500);
}

void startSendingTextBlock(int idx) {
  currentText     = textBlocks[currentMode][idx];
  currentPosition = 0;
  isSendingText   = true;
  nextTypeTime    = millis() + 50;
  updateSendingDisplay(0);
  Serial.printf("Send block %d mode %s\n", idx, modeNames[currentMode]);
}
